# TargetDeploymentResourceTypes

What Alien knows about a resource type an extension defines.

## Example Usage

```typescript
import { TargetDeploymentResourceTypes } from "@alienplatform/platform-api/models";

let value: TargetDeploymentResourceTypes = {
  controllerPlatforms: [
    "machines",
  ],
  holdsData: true,
  lifecycles: [
    "frozen",
  ],
  link: {
    permissionSets: [],
  },
  permissionSets: [],
  schemas: {
    config: {
      "key": "<value>",
      "key1": "<value>",
      "key2": "<value>",
    },
    outputs: {},
  },
  setup: {
    cloudformation: [],
    helm: [],
    importers: [
      "aws",
    ],
    terraform: [],
  },
};
```
