# VolumeRestore

A request to replace one replica's persistent volume with a volume made from a snapshot

## Example Usage

```typescript
import { VolumeRestore } from "@alienplatform/platform-api/models";

let value: VolumeRestore = {
  id: "vrst_rs1j4dytswpy7wbc819jdcx",
  deploymentId: "dep_0c29fq4a2yjb7kx3smwdgxlc",
  resourceId: "<id>",
  ordinal: 184800,
  snapshotId: "<id>",
  status: "pending",
  createdBy: {
    id: "<id>",
    kind: "serviceAccount",
  },
  createdAt: new Date("2026-04-15T04:50:05.361Z"),
  updatedAt: new Date("2026-07-03T04:17:25.608Z"),
  completedAt: new Date("2024-07-16T05:13:57.498Z"),
  cancelledAt: null,
};
```
