# DataAwsMicrovm

AWS: the image a sandbox runs from, and the lifecycle state AWS reports for it.

## Example Usage

```typescript
import { DataAwsMicrovm } from "@alienplatform/platform-api/models/operations";

let value: DataAwsMicrovm = {
  imageIdentifier: "<value>",
  status: {
    collectionIssues: [
      {
        message: "<value>",
        reason: "timed-out",
        severity: "info",
        source: "<value>",
      },
    ],
    health: "healthy",
    lifecycle: "updating",
    partial: true,
    stale: true,
  },
  backend: "awsMicrovm",
};
```
