# TerraformOutputs

Outputs from a Terraform package build.

## Example Usage

```typescript
import { TerraformOutputs } from "@alienplatform/platform-api/models";

let value: TerraformOutputs = {
  modules: {
    "key": {
      downloadUrl: "https://jittery-conservation.info/",
      filename: "example.file",
      shasum: "<value>",
      size: 266659,
      source: "<value>",
      target: "<value>",
    },
  },
  provider: {
    gpgPublicKey: {
      asciiArmor: "<value>",
      keyId: "<id>",
    },
    platforms: {},
    source: "<value>",
  },
};
```
