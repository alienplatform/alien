# DeploymentPreparedStackResourceTypes

What Alien knows about a resource type an extension defines.

## Example Usage

```typescript
import { DeploymentPreparedStackResourceTypes } from "@alienplatform/platform-api/models";

let value: DeploymentPreparedStackResourceTypes = {
  controllerPlatforms: [],
  holdsData: true,
  lifecycles: [],
  link: {
    permissionSets: [
      "<value 1>",
      "<value 2>",
      "<value 3>",
    ],
  },
  permissionSets: [
    {
      description: "ramp other second ew outdo forenenst",
      id: "<id>",
      platforms: {},
    },
  ],
  schemas: {
    config: {
      "key": "<value>",
      "key1": "<value>",
      "key2": "<value>",
    },
    outputs: {
      "key": "<value>",
    },
  },
  setup: {
    cloudformation: [
      "machines",
    ],
    helm: [
      "gcp",
    ],
    importers: [],
    terraform: [
      "machines",
    ],
  },
};
```
