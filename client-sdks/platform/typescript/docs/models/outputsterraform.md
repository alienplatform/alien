# OutputsTerraform

Outputs from a Terraform package build.

## Example Usage

```typescript
import { OutputsTerraform } from "@alienplatform/platform-api/models";

let value: OutputsTerraform = {
  modules: {},
  provider: {
    gpgPublicKey: {
      asciiArmor: "<value>",
      keyId: "<id>",
    },
    platforms: {},
    source: "<value>",
  },
  type: "terraform",
};
```
